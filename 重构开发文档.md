# SushiSwap 克隆项目 - 重构开发文档

> 生成日期: 2026-04-13
> 目标: 按照官方 SushiSwap 架构重构，实现 50+ EVM + SVM + Stellar 支持

---

## 一、项目愿景与目标

### 1.1 最终目标

| 维度 | 当前状态 | 目标状态 |
|------|---------|---------|
| EVM 链支持 | 5 条 | 50+ |
| SVM 支持 | 只有报价读取 | 完整钱包 + 交易 |
| Stellar 支持 | 无 | 基础支持 |
| 价格 API | CoinGecko 直连 | 服务端 proxy + 多层缓存 |
| 交叉链 | 无 | LI.FI 集成 |
| 类型系统 | 简单 typeof | 完善的链类型分离 |
| RPC | 公共 Llamarpc | dRPC 企业级方案 |

### 1.2 核心原则

1. **类型安全**: EVM/SVM/Stellar 类型完全分离
2. **可扩展性**: 新增链只需添加配置，无需改核心逻辑
3. **性能优先**: 服务端缓存、增量更新、批量请求
4. **用户体验**: 价格更新反馈、loading 状态、错误处理

/**

import type { EvmChainId } from "../evm"
import type { SvmChainId } from "../svm"
import type { StellarChainId } from "../stellar"

/**
 * 所有支持的链 ID 联合类型
 */
export type ChainId = EvmChainId | SvmChainId | StellarChainId

/**
 * 链类型枚举
 */
export const CHAIN_TYPE = {
  EVM: "evm",
  SVM: "svm",
  STELLAR: "stellar",
} as const

export type ChainType = (typeof CHAIN_TYPE)[keyof typeof CHAIN_TYPE]

/**
 * 获取链类型
 */
export function getChainType(chainId: ChainId): ChainType {
  if (typeof chainId === "number") return CHAIN_TYPE.EVM
  if (chainId === "pubnet" || chainId === "testnet" || chainId === "futurenet") return CHAIN_TYPE.STELLAR
  return CHAIN_TYPE.SVM
}

/**
 * 基础货币接口
 */
export interface BaseCurrency {
  address: string | null // null 表示原生货币
  symbol: string
  name: string
  decimals: number
  chainId: ChainId
  logoURI?: string
}

/**
 * EVM 货币
 */
export interface EvmCurrency extends BaseCurrency {
  chainId: EvmChainId
}

/**
 * SVM 货币
 */
export interface SvmCurrency extends BaseCurrency {
  chainId: SvmChainId
}

/**
 * Stellar 货币
 */
export interface StellarCurrency extends BaseCurrency {
  chainId: StellarChainId
}

/**
 * 统一货币类型
 */
export type Currency = EvmCurrency | SvmCurrency | StellarCurrency

/**
 * 原生货币标记接口
 */
export interface NativeCurrency {
  readonly native: true
  readonly currency: Currency
}

/**
 * 检查是否是原生货币
 */
export function isNative(currency: Currency): currency is Currency & NativeCurrency {
  return currency.address === null
}

/**
 * 检查是否是同一货币
 */
export function isSameCurrency(a: Currency, b: Currency): boolean {
  return a.chainId === b.chainId && a.address?.toLowerCase() === b.address?.toLowerCase()
}

### 4.1 dRPC 配置 (apps/web/src/lib/wagmi/config/viem.ts)

```typescript
/**
 * Wagmi + Viem 多链配置
 * 使用 dRPC 作为 RPC 提供商
 */

import { createConfig, http, fallback } from "wagmi"
import { mainnet, polygon, arbitrum, optimism, base } from "wagmi/chains"
import type { EvmChainId } from "@sushiswap/core"

// dRPC 配置
const DRPC_ID = process.env.NEXT_PUBLIC_DRPC_ID

// 链名称映射 (dRPC 使用这些名称)
const DRPC_CHAIN_NAMES: Record<number, string> = {
  [mainnet.id]: "ethereum",
  [polygon.id]: "polygon",
  [arbitrum.id]: "arbitrum",
  [optimism.id]: "optimism",
  [base.id]: "base",
  // 其他链...
}

// 创建 dRPC transport
function createDrpcTransport(chainId: EvmChainId, pollingInterval?: number) {
  const networkName = DRPC_CHAIN_NAMES[chainId]
  if (!networkName || !DRPC_ID) {
    // Fallback 到公共 RPC
    return http()
  }

  const baseUrl = `https://lb.drpc.live/ogrpc?network=${networkName}&dkey=${DRPC_ID}`

  return http(baseUrl, {
    // batch 批量请求
    batch: {
      wait: 64,
    },
    // 自定义轮询间隔
    pollingInterval: pollingInterval ?? 4000,
    // 请求超时
    timeout: 30_000,
  })
}

// 链配置
const CHAIN_CONFIGS: Record<EvmChainId, { chain: typeof mainnet; pollingInterval?: number }> = {
  [mainnet.id]: { chain: mainnet, pollingInterval: 8000 },
  [polygon.id]: { chain: polygon, pollingInterval: 4000 },
  [arbitrum.id]: { chain: arbitrum, pollingInterval: 4000 },
  [optimism.id]: { chain: optimism, pollingInterval: 2000 },
  [base.id]: { chain: base, pollingInterval: 4000 },
  // 添加更多链...
}

/**
 * 创建生产环境 wagmi config
 */
export function createProductionConfig() {
  const transports: Record<number, ReturnType<typeof createDrpcTransport>> = {}

  for (const [chainId, config] of Object.entries(CHAIN_CONFIGS)) {
    transports[Number(chainId)] = createDrpcTransport(
      Number(chainId) as EvmChainId,
      config.pollingInterval
    )
  }

  return createConfig({
    chains: Object.values(CHAIN_CONFIGS).map((c) => c.chain),
    transports,
    // 其他 wagmi 配置...
  })
}

/**
 * 创建公共(只读) wagmi config
 * 用于不需要钱包操作的页面
 */
export function createPublicConfig() {
  const transports: Record<number, ReturnType<typeof createDrpcTransport>> = {}

  for (const [chainId, config] of Object.entries(CHAIN_CONFIGS)) {
    transports[Number(chainId)] = createDrpcTransport(
      Number(chainId) as EvmChainId,
      config.pollingInterval
    )
  }

  return createConfig({
    chains: Object.values(CHAIN_CONFIGS).map((c) => c.chain),
    transports,
    // 不需要 connectors，因为是只读的
  })
}
```

**apps/web/src/lib/price/memoized.ts**

```typescript
/**
 * 价格获取 + 多层缓存
 * L1: Memory Cache (30秒)
 * L2: 后续可扩展 Redis
 */

import type { ChainId } from "@sushiswap/core"

// L1 内存缓存
interface PriceCache {
  prices: Record<string, number>
  timestamp: number
}

const memoryCache = new Map<string, PriceCache>()
const CACHE_TTL = 30_000 // 30秒

// CoinGecko 通用代币映射
const COINGECKO_IDS: Record<string, string> = {
  // ETH 链
  "1:0xc02aaa39b223fe8d0a0e5c4f27ead9083c756cc2": "wrapped-ether",
  "1:0xa0b86991c6218b36c1d19d4a2e9eb0ce3606eb48": "usd-coin",
  "1:0xdac17f958d2ee523a2206206994597c13d831ec7": "tether",
  "1:0x6b175474e89094c44da98b954eedeac495271d0f": "dai",
  "1:0x2260fac5e5542a773aa44fbcfedf7c193bc2c599": "wrapped-bitcoin",
  // SOL 链
  "mainnet:So11111111111111111111111111111111111111112": "wrapped-sol",
  "mainnet:EPjFWdd5AufqSSqeM2qN1xzybapC8G4wEGGkZwyTDt1v": "usd-coin",
  // ... 其他链
}

/**
 * 从缓存获取价格
 */
function getCachedPrices(chainId: ChainId, addresses: string[]): Record<string, number> | null {
  const cacheKey = `${chainId}`
  const cached = memoryCache.get(cacheKey)

  if (!cached) return null

  // 检查是否过期
  if (Date.now() - cached.timestamp > CACHE_TTL) {
    memoryCache.delete(cacheKey)
    return null
  }

  // 返回缓存的价格
  const result: Record<string, number> = {}
  let allFound = true

  for (const addr of addresses) {
    const key = `${chainId}:${addr.toLowerCase()}`
    if (cached.prices[key] !== undefined) {
      result[key] = cached.prices[key]
    } else {
      allFound = false
    }
  }

  return allFound ? result : null
}

/**
 * 设置缓存
 */
function setCachedPrices(chainId: ChainId, prices: Record<string, number>) {
  const cacheKey = `${chainId}`
  memoryCache.set(cacheKey, {
    prices,
    timestamp: Date.now(),
  })
}

/**
 * 获取代币价格 (带缓存)
 */
export async function getTokenPrices(
  chainId: ChainId,
  addresses: string[]
): Promise<Record<string, number>> {
  // 1. 先检查缓存
  const cached = getCachedPrices(chainId, addresses)
  if (cached) {
    return cached
  }

  // 2. 缓存没有，调用 CoinGecko
  const coinIds: string[] = []

  for (const addr of addresses) {
    const key = `${chainId}:${addr.toLowerCase()}`
    const coinId = COINGECKO_IDS[key]
    if (coinId) {
      coinIds.push(coinId)
    }
  }

  if (coinIds.length === 0) {
    return {}
  }

  // CoinGecko 请求
  const url = `https://api.coingecko.com/api/v3/simple/price?ids=${coinIds.join(",")}&vs_currencies=usd`
  const response = await fetch(url)
  const data = await response.json()

  // 3. 格式化结果并存入缓存
  const prices: Record<string, number> = {}
  let idx = 0

  for (const addr of addresses) {
    const key = `${chainId}:${addr.toLowerCase()}`
    const coinId = COINGECKO_IDS[key]
    if (coinId && data[coinId]) {
      prices[key] = data[coinId].usd
    }
    idx++
  }

  setCachedPrices(chainId, prices)

  return prices
}

/**
 * 后台刷新价格 (用于 stale-while-revalidate)
 */
export async function refreshTokenPrices(chainId: ChainId, addresses: string[]) {
  // 异步更新缓存，不阻塞请求
  getTokenPrices(chainId, addresses).catch(console.error)
}

/**
 * 批量获取代币价格
 */
export function useTokensWithCache(
  tokens: Array<{ chainId: ChainId; address: string }>,
  enabled = true
) {
  const queryClient = useQueryClient()

  // 按 chainId 分组 addresses
  const groupedTokens = useMemo(() => {
    const groups = new Map<ChainId, string[]>()
    for (const token of tokens) {
      const existing = groups.get(token.chainId) || []
      existing.push(token.address)
      groups.set(token.chainId, existing)
    }
    return groups
  }, [tokens])

  const queryKeys = useMemo(() => {
    return Array.from(groupedTokens.entries()).map(([chainId, addresses]) => ({
      chainId,
      addresses: addresses.join(","),
    }))
  }, [groupedTokens])

  const queries = useQuery({
    queryKey: ["tokens-prices", queryKeys],
    queryFn: async () => {
      const results = await Promise.all(
        Array.from(groupedTokens.entries()).map(async ([chainId, addresses]) => {
          const response = await fetch(
            `/api/price?chainId=${chainId}&addresses=${addresses.join(",")}`
          )
          return response.json()
        })
      )

      return results.reduce((acc, result) => {
        if (result.data) {
          Object.assign(acc, result.data)
        }
        return acc
      }, {} as Record<string, number>)
    },
    enabled: enabled && tokens.length > 0,
    staleTime: 900_000,
    gcTime: 86_400_000,
    refetchOnWindowFocus: false,
  })

  return {
    prices: query.data ?? {},
    isLoading: queries.isLoading,
    isError: queries.isError,
    refetch: queries.refetch,
  }
}

const lifi = new LiFi({
  integrator: "SushiSwap",
  apiKey: process.env.LIFI_API_KEY,

export async function GET(request: NextRequest) {
  const searchParams = request.nextUrl.searchParams

  const fromChain = searchParams.get("fromChain")
  const toChain = searchParams.get("toChain")
  const fromToken = searchParams.get("fromToken")
  const toToken = searchParams.get("toToken")
  const amount = searchParams.get("amount")

  if (!fromChain || !toChain || !fromToken || !toToken || !amount) {
    return NextResponse.json({ error: "Missing parameters" }, { status: 400 })
  }

  try {
    const routeOptions: RouteOptions = {
      fromChainId: Number(fromChain),
      toChainId: Number(toChain),
      fromTokenAddress: fromToken,
      toTokenAddress: toToken,
      fromAmount: amount,
      // SushiSwap 配置
      allowSwitchChain: true,
      maxPriceImpact: 0.3, // 30% 最大价格影响
    }

    // 获取路由
    const routes = await lifi.getRoutes(routeOptions)

    // 标记推荐路由
    const sortedRoutes = routes.routes.sort((a, b) => {
      // 推荐: CHEAPEST then FASTEST
      if (a.recommended && !b.recommended) return -1
      if (!a.recommended && b.recommended) return 1
      return Number(a.insuranceFee) - Number(b.insuranceFee)
    })

    return NextResponse.json(
      {
        routes: sortedRoutes,
        timestamp: Date.now(),
      },
      {
        headers: {
          "Cache-Control": "public, s-maxage=60, stale-while-revalidate=120",
        },
      }
    )
  } catch (error) {
    console.error("Cross-chain API error:", error)
    return NextResponse.json({ error: "Failed to fetch routes" }, { status: 500 })
  }

export async function POST(request: NextRequest) {
  try {
    const body = await request.json()
    const { route } = body

    if (!route) {
      return NextResponse.json({ error: "Missing route" }, { status: 400 })
    }

    // 执行交叉链交易
    const result = await lifi.executeRoute(route, {
      onTransactionStarted: (txHash) => {
        console.log("Cross-chain started:", txHash)
      },
      onTransactionCompleted: (txHash, status) => {
        console.log("Cross-chain completed:", txHash, status)
      },
    })

    return NextResponse.json({
      transactionHash: result.toolTxHash,
      status: result.status,
    })
  } catch (error) {
    console.error("Cross-chain execution error:", error)
    return NextResponse.json({ error: "Execution failed" }, { status: 500 })
  }
}
```
